```java
package cn.zcj.aether.domain.agent.service.runtime;

import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.Objects;
import java.util.concurrent.TimeUnit;

/**
 * P1-#2: LLM 响应缓存。
 * 借鉴 MetaGPT 的消息级去重 + AgentScope Java 的 Middleware 拦截模式。
 *
 * 缓存在 ModelInvoker 层透明工作——命中直接返回，未命中正常调用+写入。
 * Key = modelName + ":" + messages 的内容哈希
 * Value = ModelInvoker.ModelCallResult
 */
@Slf4j
@Component
public class ModelCallCache {

    private final Cache<String, ModelInvoker.ModelCallResult> cache;

    public ModelCallCache() {
        this.cache = Caffeine.newBuilder()
                .maximumSize(1000)
                .expireAfterWrite(60, TimeUnit.SECONDS)
                .recordStats()
                .build();
    }

    /**
     * 生成缓存 key
     */
    public static String cacheKey(String modelName, List<org.springframework.ai.chat.messages.Message> messages) {
        int hash = Objects.hash(modelName, messages.stream()
                .map(m -> m.getText() != null ? m.getText() : "")
                .toList());
        return modelName + ":" + hash;
    }

    /**
     * 查询缓存
     */
    public ModelInvoker.ModelCallResult get(String key) {
        ModelInvoker.ModelCallResult result = cache.getIfPresent(key);
        if (result != null) {
            log.debug("LLM 缓存命中: key={}", key);
        }
        return result;
    }

    /**
     * 写入缓存（仅缓存成功的非 error 响应）
     */
    public void put(String key, ModelInvoker.ModelCallResult result, int ttlSeconds) {
        if (result.hasError()) return; // 不缓存错误响应
        cache.policy().expireVariably().ifPresent(ev -> {
            // 使用可变 TTL
        });
        cache.put(key, result);
        log.debug("LLM 缓存写入: key={}", key);
    }

    /**
     * 清空全部缓存
     */
    public void clear() {
        cache.invalidateAll();
        log.info("LLM 缓存已清空");
    }

    /**
     * 缓存命中率（监控用）
     */
    public double hitRate() {
        return cache.stats().hitRate();
    }
}
```

```java
    /**
     * P1-#2: 带缓存的流式调用。
     * 缓存命中 → 直接返回；缓存未命中 → callWithStream → 写入缓存。
     */
    public ModelCallResult callWithStreamCached(
            ChatModel chatModel,
            List<Message> messages,
            String systemPrompt,
            String modelName,
            boolean cacheEnabled,
            int cacheTtlSeconds) {

        if (!cacheEnabled) {
            return callWithStream(chatModel, messages, systemPrompt, modelName);
        }

        String key = ModelCallCache.cacheKey(modelName, messages);
        ModelCallResult cached = modelCallCache.get(key);
        if (cached != null) {
            return ModelCallResult.builder()
                    .events(cached.getEvents())
                    .fullText(cached.getFullText())
                    .toolCalls(cached.getToolCalls())
                    .inputTokens(cached.getInputTokens())
                    .outputTokens(cached.getOutputTokens())
                    .cacheTokens(cached.getOutputTokens())  // 缓存命中 = 节省了 output tokens
                    .build();
        }

        ModelCallResult result = callWithStream(chatModel, messages, systemPrompt, modelName);
        if (!result.hasError()) {
            modelCallCache.put(key, result, cacheTtlSeconds);
        }
        return result;
    }
```

```java
package cn.zcj.aether.domain.agent.service.agent.core;

import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.Data;
import lombok.NoArgsConstructor;

import java.util.List;

/**
 * P1-#4: Agent 执行计划。
 * 借鉴 MetaGPT 的 Plan 模型。
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class Plan {
    /** 原始用户任务描述 */
    private String taskDescription;
    /** 执行步骤列表 */
    private List<Step> steps;
    /** 步骤总数 */
    private int totalSteps;

    @Data
    @Builder
    @NoArgsConstructor
    @AllArgsConstructor
    public static class Step {
        /** 步骤编号（从 1 开始） */
        private int id;
        /** 步骤描述 */
        private String description;
        /** 预期产出 */
        private String expectedOutput;
        /** 执行状态 */
        @Builder.Default
        private String status = "pending";  // pending | running | completed | failed
        /** 执行结果 */
        private String result;
    }

    public static Plan parse(String jsonText) {
        try {
            var mapper = new com.fasterxml.jackson.databind.ObjectMapper();
            var root = mapper.readTree(jsonText);
            var stepsNode = root.get("steps");
            List<Step> steps = new java.util.ArrayList<>();
            if (stepsNode != null && stepsNode.isArray()) {
                int idx = 1;
                for (var node : stepsNode) {
                    steps.add(Step.builder()
                            .id(idx++)
                            .description(node.has("description") ? node.get("description").asText() : "")
                            .expectedOutput(node.has("expectedOutput") ? node.get("expectedOutput").asText() : "")
                            .build());
                }
            }
            return Plan.builder()
                    .taskDescription(root.has("taskDescription") ? root.get("taskDescription").asText() : "")
                    .steps(steps)
                    .totalSteps(steps.size())
                    .build();
        } catch (Exception e) {
            throw new RuntimeException("无法解析 LLM 返回的计划 JSON: " + jsonText, e);
        }
    }
}
```

```java
package cn.zcj.aether.domain.agent.service.agent.impl;

import cn.zcj.aether.domain.agent.service.agent.checkpoint.CheckpointCollector;
import cn.zcj.aether.domain.agent.service.agent.core.*;
import cn.zcj.aether.domain.agent.service.agent.hook.AgentHook;
import cn.zcj.aether.domain.agent.service.agent.middleware.MiddlewareChain;
import cn.zcj.aether.domain.agent.service.agent.middleware.impl.DefaultMiddlewareChain;
import cn.zcj.aether.domain.agent.service.context.ContextManager;
import cn.zcj.aether.domain.agent.service.event.AgentEventPublisher;
import cn.zcj.aether.domain.agent.service.runtime.ModelInvoker;
import cn.zcj.aether.domain.agent.service.runtime.RuntimeEvent;
import cn.zcj.aether.domain.agent.service.tool.ToolExecutor;
import io.reactivex.rxjava3.core.BackpressureStrategy;
import io.reactivex.rxjava3.core.Flowable;
import io.reactivex.rxjava3.core.FlowableEmitter;
import lombok.Getter;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.chat.model.ChatModel;
import org.springframework.ai.chat.messages.SystemMessage;
import org.springframework.ai.chat.messages.UserMessage;
import org.springframework.ai.chat.prompt.Prompt;

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.atomic.AtomicBoolean;

/**
 * P1-#4: Plan-Act-Synthesize 三阶段 Agent。
 * 借鉴 MetaGPT 的 PLAN_AND_ACT 模式：
 *   Phase 1（Plan）: LLM 将任务分解为 JSON 步骤计划
 *   Phase 2（Act）:   逐步执行，每步独立 ReAct 子循环
 *   Phase 3（Synthesize）: 合并所有步骤结果，输出最终答案
 *
 * 借鉴 AutoGen MagenticOne 的 Orchestrator 模式——PlanActAgent 作为编排器
 * 协调内部 ReActAgent 子循环执行每个步骤。
 */
@Slf4j
public class PlanActAgent extends BaseAgent {

    private final ChatModel chatModel;
    private final ModelInvoker modelInvoker;
    private final ToolExecutor toolExecutor;
    private final ContextManager contextManager;
    @Getter
    private final List<AgentHook> hooks = new ArrayList<>();
    private final AgentEventPublisher eventPublisher;
    private final CheckpointCollector checkpointCollector;

    public PlanActAgent(AgentConfig config,
                         ChatModel chatModel,
                         ModelInvoker modelInvoker,
                         ToolExecutor toolExecutor,
                         ContextManager contextManager,
                         AgentEventPublisher eventPublisher,
                         CheckpointCollector checkpointCollector) {
        super(config);
        this.chatModel = chatModel;
        this.modelInvoker = modelInvoker;
        this.toolExecutor = toolExecutor;
        this.contextManager = contextManager;
        this.eventPublisher = eventPublisher;
        this.checkpointCollector = checkpointCollector;
    }

    @Override
    public Flowable<RuntimeEvent> execute(RuntimeContext ctx) {
        return Flowable.create(emitter -> {
            try {
                MiddlewareChain chain = new DefaultMiddlewareChain();
                AtomicBoolean aborted = new AtomicBoolean(false);
                emitter.setCancellable(() -> aborted.set(true));

                // ====== Phase 1: Plan（规划） ======
                log.info("[PlanActAgent] Phase 1: Plan — 生成执行计划");
                Plan plan = generatePlan(ctx.message(), config.getInstruction());

                if (plan.getSteps().isEmpty()) {
                    emitter.onNext(RuntimeEvent.error("无法生成执行计划"));
                    emitter.onComplete();
                    return;
                }

                emitter.onNext(RuntimeEvent.text("📋 执行计划（共 " + plan.getTotalSteps() + " 步）:\n"));
                for (var step : plan.getSteps()) {
                    emitter.onNext(RuntimeEvent.text(
                            "  " + step.getId() + ". " + step.getDescription() + "\n"));
                }

                if (aborted.get()) return;

                // ====== Phase 2: Act（逐步执行） ======
                log.info("[PlanActAgent] Phase 2: Act — 逐步执行");
                List<StepResult> stepResults = new ArrayList<>();

                for (var step : plan.getSteps()) {
                    if (aborted.get()) break;

                    step.setStatus("running");
                    long stepStart = System.currentTimeMillis();
                    emitter.onNext(RuntimeEvent.text(
                            "\n▶ 步骤 " + step.getId() + "/" + plan.getTotalSteps()
                            + ": " + step.getDescription() + "\n"));

                    // 为每个步骤创建临时的子 Agent 配置
                    AgentConfig stepConfig = AgentConfig.builder()
                            .name(config.getName() + "-step-" + step.getId())
                            .instruction(step.getDescription() + "\n预期产出: " + step.getExpectedOutput())
                            .toolNames(config.getToolNames())
                            .modelRef(config.getModelRef())
                            .agentType("react")
                            .checkpointEnabled(false)  // 子循环不独立保存检查点
                            .build();

                    ReActAgent subAgent = new ReActAgent(stepConfig, chatModel, modelInvoker,
                            toolExecutor, contextManager, eventPublisher, checkpointCollector);

                    // 执行子循环并收集输出
                    RuntimeContext stepCtx = new RuntimeContext(
                            ctx.userId(), ctx.sessionId() + "-s" + step.getId(),
                            step.getDescription(), ctx.correlationId(), ctx.metadata());

                    List<String> outputs = new ArrayList<>();
                    subAgent.execute(stepCtx)
                            .doOnNext(event -> {
                                emitter.onNext(event);
                                if (event.getType() == RuntimeEvent.EventType.textDelta
                                        && event.getText() != null) {
                                    outputs.add(event.getText());
                                }
                            })
                            .doOnComplete(() -> {
                                String output = String.join("", outputs);
                                stepResults.add(StepResult.builder()
                                        .stepId(step.getId())
                                        .description(step.getDescription())
                                        .output(output)
                                        .success(true)
                                        .durationMs(System.currentTimeMillis() - stepStart)
                                        .build());
                            })
                            .blockingSubscribe();

                    step.setStatus("completed");
                    step.setResult(stepResults.get(stepResults.size() - 1).getOutput());
                }

                // ====== Phase 3: Synthesize（合成） ======
                log.info("[PlanActAgent] Phase 3: Synthesize — 合并结果");
                String finalAnswer = synthesize(plan, stepResults, config.getInstruction());
                emitter.onNext(RuntimeEvent.text(finalAnswer));
                emitter.onNext(RuntimeEvent.done());
                emitter.onComplete();

            } catch (Exception e) {
                log.error("[PlanActAgent] 执行失败", e);
                emitter.onNext(RuntimeEvent.error(e.getMessage()));
                emitter.onComplete();
            }
        }, BackpressureStrategy.BUFFER);
    }

    // ====== Phase 1: Generate Plan ======
    private Plan generatePlan(String userMessage, String instruction) {
        String prompt = String.format("""
                你是一个任务规划助手。请将以下任务分解为具体可执行的步骤。
                严格返回 JSON 格式，不要其他内容。

                { "taskDescription": "原始任务描述", "steps": [
                  { "description": "步骤的具体执行描述", "expectedOutput": "该步骤预期产生的产出" }
                ]}

                任务：%s
                上下文：%s
                """, userMessage, instruction != null ? instruction : "");

        var response = chatModel.call(new Prompt(new UserMessage(prompt)));
        String text = response.getResult().getOutput().getText();
        return Plan.parse(extractJson(text));
    }

    // ====== Phase 3: Synthesize ======
    private String synthesize(Plan plan, List<StepResult> results, String instruction) {
        StringBuilder context = new StringBuilder("请根据以下步骤的执行结果，生成最终的综合回答。\n\n");
        context.append("原始任务：").append(plan.getTaskDescription()).append("\n\n");
        context.append("步骤执行结果：\n");
        for (var r : results) {
            context.append("步骤").append(r.getStepId()).append(": ").append(r.getDescription()).append("\n");
            context.append("结果: ").append(r.getOutput()).append("\n\n");
        }
        context.append("系统指令：").append(instruction != null ? instruction : "综合所有步骤结果，给出最终答案");

        var response = chatModel.call(new Prompt(new SystemMessage(context.toString())));
        return response.getResult().getOutput().getText();
    }

    private String extractJson(String text) {
        if (text == null) return "{}";
        int start = text.indexOf('{');
        int end = text.lastIndexOf('}');
        if (start >= 0 && end > start) {
            return text.substring(start, end + 1);
        }
        return text;
    }
}
```

```java
package cn.zcj.aether.domain.agent.service.agent.permission;

import lombok.extern.slf4j.Slf4j;

import java.util.Collections;
import java.util.Map;
import java.util.Set;
import java.util.concurrent.ConcurrentHashMap;

/**
 * P1-#9: 工具白名单/黑名单规则。
 * 借鉴 AgentScope Java 的 PermissionRule 链式评估 + cc-haha 的显式 allow/deny 列表。
 *
 * 优先级顺序: 显式 allowlist（白名单）= 放行 > denylist（黑名单）= 拒绝 > 默认放行
 */
@Slf4j
public class ToolAllowlistRule implements PermissionRule {

    private final Set<String> allowlist;
    private final Set<String> denylist;

    public ToolAllowlistRule() {
        this(Set.of(), Set.of());
    }

    public ToolAllowlistRule(Set<String> allowlist, Set<String> denylist) {
        this.allowlist = ConcurrentHashMap.newKeySet();
        this.denylist = ConcurrentHashMap.newKeySet();
        this.allowlist.addAll(allowlist);
        this.denylist.addAll(denylist);
    }

    @Override
    public PermissionDecision evaluate(String toolName, Map<String, Object> toolArgs,
                                        PermissionMode mode, int toolCallCount) {
        // BYPASS 模式 → 跳过
        if (mode == PermissionMode.BYPASS) return null;

        // 显式黑名单 → 拒绝
        if (!denylist.isEmpty() && denylist.contains(toolName)) {
            log.warn("工具 [{}] 在执行黑名单中，已拒绝", toolName);
            return PermissionDecision.deny("工具 [" + toolName + "] 在执行黑名单中");
        }

        // 显式白名单配置且工具不在列表中 → 拒绝
        if (!allowlist.isEmpty() && !allowlist.contains(toolName)) {
            log.warn("工具 [{}] 未在执行白名单中，已拒绝", toolName);
            return PermissionDecision.deny("工具 [" + toolName + "] 未在执行白名单中，允许的工具: " + allowlist);
        }

        return null; // 不干预，交由下一条规则
    }

    @Override
    public int priority() { return 15; }

    public void setAllowlist(Set<String> tools) {
        this.allowlist.clear();
        this.allowlist.addAll(tools);
    }

    public void setDenylist(Set<String> tools) {
        this.denylist.clear();
        this.denylist.addAll(tools);
    }
}
```

```java
package cn.zcj.aether.domain.agent.service.agent.permission;

import lombok.extern.slf4j.Slf4j;

import java.util.Set;

/**
 * P1-#9: 敏感参数脱敏规则。
 * 检测工具参数名包含 apiKey/password/secret/token → 替换为 "***"。
 * 借鉴 AgentScope Java 的 PermissionMiddleware 参数改写模式。

@Slf4j
public class SensitiveArgMaskRule implements PermissionRule {

    private static final Set<String> SENSITIVE_KEYS = Set.of(
            "apikey", "api_key", "api-key", "password", "passwd",
            "secret", "token", "access_token", "credential", "private_key");

    @Override
    public PermissionDecision evaluate(String toolName, Map<String, Object> toolArgs,
                                        PermissionMode mode, int toolCallCount) {
        if (mode == PermissionMode.BYPASS) return null;
        if (toolArgs == null || toolArgs.isEmpty()) return null;

        boolean masked = false;
        for (String key : toolArgs.keySet()) {
            String lower = key.toLowerCase().replace("-", "").replace("_", "");
            if (SENSITIVE_KEYS.contains(lower) && toolArgs.get(key) instanceof String val && !val.isEmpty()) {
                toolArgs.put(key, "***");
                masked = true;
            }
        }

        if (masked) {
            log.debug("工具 [{}] 参数已脱敏", toolName);
        }
        return null; // 不干预执行决策，仅改写参数
    }

    @Override
    public int priority() { return 5; }
}
```

### Task 9: P1-#1 — ModelInvoker 新增 callWithStreamAsync()

**Files:**
- Modify: `aether-domain/src/main/java/cn/zcj/aether/domain/agent/service/runtime/ModelInvoker.java`

- [ ] **Step 1: 新增异步调用方法**

在 `callWithStream()` 之后新增：
```java
    /**
     * P1-#1: 异步流式调用 —— 返回 Mono 而非 blocking。
     * 借鉴 AgentScope Java 的 Mono.defer() 惰性求值 + Flux.collectList() 非阻塞收集。
     *
     * 调用方可以在上层组合 Mono，例如用 Mono.zip() 并行调用多个 LLM。
     */
    public reactor.core.publisher.Mono<ModelCallResult> callWithStreamAsync(
            ChatModel chatModel,
            List<Message> messages,
            String systemPrompt,
            String modelName) {

        log.info("异步模型调用: model={}, messagesCount={}", modelName, messages.size());

        List<Message> fullMessages = new ArrayList<>();
        fullMessages.add(new org.springframework.ai.chat.messages.SystemMessage(systemPrompt));
        fullMessages.addAll(messages);

        Prompt prompt = new Prompt(fullMessages);

        return chatModel.stream(prompt)
                .collectList()
                .map(responses -> {
                    List<RuntimeEvent> events = new ArrayList<>();
                    StringBuilder fullText = new StringBuilder();
                    List<ToolCallDef> toolCalls = new ArrayList<>();

                    if (responses != null) {
                        for (org.springframework.ai.chat.model.ChatResponse response : responses) {
                            var generations = response.getResults();
                            if (generations == null) continue;

                            for (var gen : generations) {
                                var output = gen.getOutput();
                                if (output == null) continue;

                                String text = output.getText();
                                if (text != null && !text.isEmpty()) {
                                    fullText.append(text);
                                    events.add(RuntimeEvent.text(text));
                                }

                                var tcList = output.getToolCalls();
                                if (tcList != null && !tcList.isEmpty()) {
                                    for (var tc : tcList) {
                                        Map<String, Object> parsedArgs = parseArguments(tc.arguments());
                                        toolCalls.add(ToolCallDef.builder()
                                                .id(tc.id()).name(tc.name()).input(parsedArgs).build());
                                        events.add(RuntimeEvent.builder()
                                                .type(RuntimeEvent.EventType.toolCall)
                                                .toolCallId(tc.id()).toolName(tc.name())
                                                .toolInput(tc.arguments()).build());
                                    }
                                }
                            }
                        }
                    }

                    log.info("异步模型调用完成: model={}, textLength={}, toolCalls={}",
                            modelName, fullText.length(), toolCalls.size());

                    return ModelCallResult.builder()
                            .events(events)
                            .fullText(fullText.toString())
                            .toolCalls(toolCalls)
                            .build();
                });
    }
```
